/**
 * submitLeadValidator.ts — High Tier Only
 * Validates the Submit Outbound Lead form before adding to queue.
 * Called by the Apps Script AND by the backend API.
 */

import {
  ALLOWED_RELATIONSHIP_TYPES,
  BLOCKED_RELATIONSHIP_TYPES,
  ALLOWED_CONSENT_STATUSES,
  BLOCKED_CONSENT_STATUSES,
  ALLOWED_CAMPAIGN_TYPES,
  isValidPhone,
} from './outboundEligibility';

export interface SubmitLeadForm {
  leadName: string;
  phoneNumber: string;
  email?: string;
  campaignType: string;
  serviceInterest?: string;
  desiredOutcome: string;
  relationshipType: string;
  consentStatus: string;
  consentSource?: string;
  lastInteractionDate?: string;
  preferredCallWindow?: string;
  notes?: string;
  clientAttestation: boolean;
}

export interface ValidationResult {
  valid: boolean;
  errors: string[];
}

/**
 * Validate a submitted outbound lead form.
 * Returns validation result with all errors listed (not fail-fast).
 */
export function validateSubmitLeadForm(formData: SubmitLeadForm): ValidationResult {
  const errors: string[] = [];

  // ── Required fields ─────────────────────────────────────────────────────────
  if (!formData.leadName?.trim()) {
    errors.push('Lead Name is required');
  }

  if (!formData.phoneNumber?.trim()) {
    errors.push('Phone Number is required');
  } else if (!isValidPhone(formData.phoneNumber)) {
    errors.push('Phone Number is invalid — must be 10+ digits');
  }

  if (!formData.campaignType) {
    errors.push('Campaign Type is required');
  }

  if (!formData.desiredOutcome?.trim()) {
    errors.push('Desired Outcome is required — describe what you want the AI to achieve');
  }

  if (!formData.relationshipType) {
    errors.push('Relationship Type is required');
  }

  if (!formData.consentStatus) {
    errors.push('Consent Status is required');
  }

  if (!formData.clientAttestation) {
    errors.push('Client Attestation must be confirmed — you are certifying this is a permitted contact');
  }

  // ── Allowed list checks ─────────────────────────────────────────────────────
  if (formData.campaignType && !ALLOWED_CAMPAIGN_TYPES.includes(formData.campaignType)) {
    errors.push(
      `Campaign Type "${formData.campaignType}" is not permitted. ` +
      `Allowed: ${ALLOWED_CAMPAIGN_TYPES.join(', ')}`
    );
  }

  if (formData.relationshipType && !ALLOWED_RELATIONSHIP_TYPES.includes(formData.relationshipType)) {
    errors.push(
      `Relationship Type "${formData.relationshipType}" is not permitted.`
    );
  }

  // ── Blocked list checks (explicit hard blocks) ──────────────────────────────
  if (formData.relationshipType && BLOCKED_RELATIONSHIP_TYPES.includes(formData.relationshipType)) {
    errors.push(
      `Relationship Type "${formData.relationshipType}" is BLOCKED. ` +
      `Scraped, purchased, or unknown-source leads cannot be called.`
    );
  }

  if (formData.consentStatus && BLOCKED_CONSENT_STATUSES.includes(formData.consentStatus)) {
    errors.push(
      `Consent Status "${formData.consentStatus}" is BLOCKED. ` +
      `Do Not Call and unknown-consent leads cannot be contacted.`
    );
  }

  if (formData.consentStatus && !ALLOWED_CONSENT_STATUSES.includes(formData.consentStatus) &&
      !BLOCKED_CONSENT_STATUSES.includes(formData.consentStatus)) {
    errors.push(
      `Consent Status "${formData.consentStatus}" is unrecognized. ` +
      `Allowed: ${ALLOWED_CONSENT_STATUSES.join(', ')}`
    );
  }

  // ── Desired outcome quality check ───────────────────────────────────────────
  if (formData.desiredOutcome && formData.desiredOutcome.trim().length < 10) {
    errors.push('Desired Outcome is too short — please describe what you want the AI to accomplish');
  }

  return {
    valid: errors.length === 0,
    errors,
  };
}

/**
 * Sanitize form data before writing to sheet.
 * Trims strings, normalizes phone format.
 */
export function sanitizeFormData(formData: SubmitLeadForm): SubmitLeadForm {
  return {
    ...formData,
    leadName:           formData.leadName?.trim() ?? '',
    phoneNumber:        normalizePhone(formData.phoneNumber ?? ''),
    email:              formData.email?.trim().toLowerCase() ?? '',
    desiredOutcome:     formData.desiredOutcome?.trim() ?? '',
    notes:              formData.notes?.trim() ?? '',
    consentSource:      formData.consentSource?.trim() ?? '',
    preferredCallWindow: formData.preferredCallWindow?.trim() ?? '',
  };
}

function normalizePhone(phone: string): string {
  const digits = phone.replace(/\D/g, '');
  if (digits.length === 10) return `+1${digits}`;
  if (digits.length === 11 && digits.startsWith('1')) return `+${digits}`;
  return phone; // return as-is if we can't normalize
}
