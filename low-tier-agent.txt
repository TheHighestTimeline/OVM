/**
 * staffAssigner.ts — Medium / High Tier
 * Assigns calls to staff members based on service and/or location.
 * Reads Staff Assignment Rules from the client workbook.
 */

import { readSheetTab, updateSheetCell } from './googleSheets';

export interface CallData {
  callId: string;
  businessId: string;
  serviceRequested: string;
  callerPhone: string;
  urgency: string;
  location?: string;
}

export interface StaffAssignmentRule {
  service_name: string;
  location: string;
  assigned_staff_name: string;
  assigned_staff_email: string;
  assigned_staff_phone: string;
  escalation_priority: string;   // 'Low' | 'Normal' | 'High' | 'Urgent'
  notes: string;
}

export interface AssignmentResult {
  staffName: string | null;
  staffEmail: string | null;
  staffPhone: string | null;
  escalationPriority: string;
  matchedOn: 'service' | 'location' | 'both' | 'default' | 'none';
}

/**
 * Determine which staff member should handle this call.
 * Returns null if no matching rule found.
 *
 * Staff Assignment Rules tab columns:
 * service_name | location | assigned_staff_name | assigned_staff_email |
 * assigned_staff_phone | escalation_priority | notes
 */
export async function assignStaff(
  callData: CallData,
  sheetId: string
): Promise<AssignmentResult> {
  const noAssignment: AssignmentResult = {
    staffName: null,
    staffEmail: null,
    staffPhone: null,
    escalationPriority: 'Normal',
    matchedOn: 'none',
  };

  try {
    const rows = await readSheetTab(sheetId, 'Staff Assignment Rules');
    if (!rows || rows.length < 2) return noAssignment;

    const rules = parseRulesFromSheet(rows);
    const match = findBestMatch(rules, callData);

    if (!match) {
      console.log(`[staffAssigner] No staff rule matched for service="${callData.serviceRequested}"`);
      return noAssignment;
    }

    console.log(`[staffAssigner] Assigned "${match.rule.assigned_staff_name}" (matched on: ${match.matchedOn})`);

    return {
      staffName:         match.rule.assigned_staff_name,
      staffEmail:        match.rule.assigned_staff_email,
      staffPhone:        match.rule.assigned_staff_phone,
      escalationPriority: match.rule.escalation_priority || 'Normal',
      matchedOn:         match.matchedOn,
    };
  } catch (err) {
    console.error('[staffAssigner] Error loading staff rules:', err);
    return noAssignment;
  }
}

// ── Internal helpers ──────────────────────────────────────────────────────────

function parseRulesFromSheet(rows: string[][]): StaffAssignmentRule[] {
  return rows.slice(1)
    .filter(row => row[0] || row[1]) // skip blank rows
    .map(row => ({
      service_name:          (row[0] ?? '').toLowerCase().trim(),
      location:              (row[1] ?? '').toLowerCase().trim(),
      assigned_staff_name:   row[2] ?? '',
      assigned_staff_email:  row[3] ?? '',
      assigned_staff_phone:  row[4] ?? '',
      escalation_priority:   row[5] ?? 'Normal',
      notes:                 row[6] ?? '',
    }));
}

interface MatchResult {
  rule: StaffAssignmentRule;
  matchedOn: 'both' | 'service' | 'location' | 'default';
  score: number;
}

function findBestMatch(
  rules: StaffAssignmentRule[],
  callData: CallData
): MatchResult | null {
  const service  = (callData.serviceRequested ?? '').toLowerCase();
  const location = (callData.location ?? '').toLowerCase();

  const matches: MatchResult[] = [];

  for (const rule of rules) {
    const serviceMatch  = rule.service_name  && service.includes(rule.service_name);
    const locationMatch = rule.location      && location.includes(rule.location);
    const isDefault     = !rule.service_name && !rule.location; // catch-all rule

    if (serviceMatch && locationMatch) {
      matches.push({ rule, matchedOn: 'both', score: 3 });
    } else if (serviceMatch) {
      matches.push({ rule, matchedOn: 'service', score: 2 });
    } else if (locationMatch) {
      matches.push({ rule, matchedOn: 'location', score: 1 });
    } else if (isDefault) {
      matches.push({ rule, matchedOn: 'default', score: 0 });
    }
  }

  if (matches.length === 0) return null;

  // Return highest-score match
  return matches.sort((a, b) => b.score - a.score)[0];
}
